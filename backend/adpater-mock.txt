func loadDB() error { // only load the database has not been loaded to tmp yet
	if _, err := os.Stat(tmpDBPath); err == nil {
		fmt.Println("Database file already exists, skipping download")
		return nil
	}
	fmt.Println("Downloading database from Google Cloud Storage...")
	ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
	defer cancel()
	client, err := storage.NewClient(ctx)
	if err != nil {
		return fmt.Errorf("storage.NewClient: %v", err)
	}
	defer client.Close()

	rc, err := client.Bucket(bucketName).Object(db).NewReader(ctx)
	if err != nil {
		return fmt.Errorf("unable to read object: %v", err)
	}
	defer rc.Close()

	outFile, err := os.Create(tmpDBPath)
	if err != nil {
		return fmt.Errorf("os.Create: %v", err)
	}
	defer outFile.Close()

	if _, err := io.Copy(outFile, rc); err != nil {
		return fmt.Errorf("copying DB file: %v", err)
	}
	fmt.Println("Database downloaded successfully")
	return nil
}

func updateDB() error {
	cloudMutex.Lock()
	defer cloudMutex.Unlock()

	fmt.Println("Update database to Google Cloud Storage...")

	ctx := context.Background()
	client, err := storage.NewClient(ctx)
	if err != nil {
		return fmt.Errorf("storage.NewClient: %w", err)
	}
	defer client.Close()

	f, err := os.Open(tmpDBPath)
	if err != nil {
		return fmt.Errorf("os.Open (%s): %w", tmpDBPath, err)
	}
	defer f.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 50*time.Second)
	defer cancel()

	wc := client.Bucket(bucketName).Object(db).NewWriter(ctx)
	if _, err = io.Copy(wc, f); err != nil {
		return fmt.Errorf("io.Copy: %w", err)
	}
	if err := wc.Close(); err != nil {
		return fmt.Errorf("Writer.Close: %w", err)
	}
	fmt.Fprintf(os.Stdout, "Successfully uploaded %s to bucket %s as %s\n", tmpDBPath, bucketName, db)
	return nil
}

func saveHintClicked(teamID int, questionID int, clicked int) error {
	dbMutex.Lock()
	defer dbMutex.Unlock()

	fmt.Printf("Saving hint_clicked for team ID: %d at station ID: %d, clicked = %d\n", teamID, questionID, clicked)
	if err := loadDB(); err != nil {
		return fmt.Errorf("failed to load database: %v", err)
	}
	db, err := sql.Open("sqlite3", tmpDBPath)
	if err != nil {
		return fmt.Errorf("open DB: %v", err)
	}
	defer db.Close()

	if err := db.Ping(); err != nil {
		return fmt.Errorf("database ping failed: %v", err)
	}
	currentClicked, err := loadHintClicked(teamID, questionID)
	if err != nil {
		return fmt.Errorf("load hint clicked failed: %v", err)
	}
	if currentClicked < clicked {
		res, err := db.Exec("UPDATE hint_click SET clicked = ? WHERE team_id = ? AND question_id = ?", clicked, teamID, questionID)
		if err != nil {
			return fmt.Errorf("update hint_click failed: %v", err)
		}
		rows, err := res.RowsAffected()
		if err != nil {
			return fmt.Errorf("checking rows affected: %v", err)
		}
		if rows == 0 {
			fmt.Printf("No existing record found for team %d at station %d, inserting new record\n", teamID, questionID)
			_, err := db.Exec("INSERT INTO hint_click (team_id, question_id, clicked) VALUES (?, ?, ?)", teamID, questionID, clicked)
			if err != nil {
				return fmt.Errorf("insert hint_click failed: %v", err)
			}
			fmt.Println("Successfully inserted new record for hint_click.")
		}

		if err := updateDB(); err != nil {
			fmt.Printf("Warning: Failed to sync database to cloud: %v\n", err)
		}
		fmt.Println("Successfully saved to hint_click.")
		return nil
	} else {
		fmt.Printf("No update needed, current clicked (%d) is greater than or equal to new clicked (%d)\n", currentClicked, clicked)
		return nil
	}
}

// Queue manager

func addQueue(teamID int, questionID int) (bool, error) { // recheck this func
	dbMutex.Lock()
	defer dbMutex.Unlock()

	fmt.Printf("Adding team %d to queues for question %d\n", teamID, questionID)
	if err := loadDB(); err != nil {
		return false, fmt.Errorf("failed to load database: %v", err)
	}

	db, err := sql.Open("sqlite3", tmpDBPath)
	if err != nil {
		return false, fmt.Errorf("open DB: %v", err)
	}
	defer db.Close()

	if err := db.Ping(); err != nil {
		return false, fmt.Errorf("database ping failed: %v", err)
	}

	var existingPosition int
	err = db.QueryRow(`SELECT position FROM queues WHERE queue_id = ? AND team_id = ?`, questionID, teamID).Scan(&existingPosition)
	if err == nil {
		fmt.Printf("Team %d is already in queue for question %d at position %d\n", teamID, questionID, existingPosition)
		return true, nil // Team is already in queue, return success
	} else if err != sql.ErrNoRows {
		return false, fmt.Errorf("failed to check existing queue position: %v", err)
	}

	var count int
	err = db.QueryRow(`SELECT COUNT(*) FROM queues WHERE queue_id = ?`, questionID).Scan(&count)
	if err != nil {
		return false, fmt.Errorf("failed to count queue: %v", err)
	}

	if count > 1 {
		fmt.Println("Queue is full, cannot add more teams.")
		return false, nil
	}

	position := count + 1

	_, err = db.Exec(`
		INSERT INTO queues (queue_id, position, team_id)
		VALUES (?, ?, ?)`, questionID, position, teamID)
	if err != nil {
		return false, fmt.Errorf("insert into queues: %v", err)
	}
	if err := updateDB(); err != nil {
		fmt.Printf("Warning: Failed to sync database to cloud: %v\n", err)
	}
	fmt.Println("Successfully added to queue at position", position)
	return true, nil
}

func removeQueue(teamID int, questionID int) error {
	dbMutex.Lock()
	defer dbMutex.Unlock()

	fmt.Printf("Removing team %d from queue for question %d\n", teamID, questionID)

	if err := loadDB(); err != nil {
		return fmt.Errorf("failed to load database: %v", err)
	}

	db, err := sql.Open("sqlite3", tmpDBPath)
	if err != nil {
		return fmt.Errorf("open DB: %v", err)
	}
	defer db.Close()

	if err := db.Ping(); err != nil {
		return fmt.Errorf("database ping failed: %v", err)
	}

	var pos int
	err = db.QueryRow(`SELECT position FROM queues WHERE queue_id = ? AND team_id = ?`, questionID, teamID).Scan(&pos)
	if err == sql.ErrNoRows {
		fmt.Println("Team not found in queue, nothing to remove.")
		return nil
	} else if err != nil {
		return fmt.Errorf("failed to find team position: %v", err)
	}
	_, err = db.Exec(`DELETE FROM queues WHERE queue_id = ? AND team_id = ?`, questionID, teamID)
	if err != nil {
		return fmt.Errorf("failed to delete team from queue: %v", err)
	}

	_, err = db.Exec(`
		UPDATE queues
		SET position = position - 1
		WHERE queue_id = ? AND position > ?
	`, questionID, pos)
	if err != nil {
		return fmt.Errorf("failed to shift queue: %v", err)
	}
	if err := updateDB(); err != nil {
		fmt.Printf("Warning: Failed to sync database to cloud: %v\n", err)
	}
	fmt.Println("Successfully removed team from queue and updated positions.")
	return nil
}
func saveAndLoadLoginTime(teamID int, loginTime time.Time) (time.Time, error) {
	dbMutex.Lock()
	defer dbMutex.Unlock()

    if err := loadDB(); err != nil {
        return time.Time{}, fmt.Errorf("failed to load database: %v", err)
    }

    db, err := sql.Open("sqlite3", tmpDBPath)
    if err != nil {
        return time.Time{}, fmt.Errorf("open DB: %v", err)
    }
    defer db.Close()

    _, err = db.Exec("UPDATE teams SET login_time = ? WHERE team_id = ? AND login_time IS NULL",
        loginTime.Format(time.RFC3339), teamID)
    if err != nil {
        return time.Time{}, fmt.Errorf("update login time: %v", err)
    }

	fmt.Println("Successfully updated login time for team:", teamID)
	var existingLoginTime string
	err = db.QueryRow("SELECT login_time FROM teams WHERE team_id = ?", teamID).Scan(&existingLoginTime);
	if err != nil {
		return time.Time{}, fmt.Errorf("query login time: %v", err)
	}
	var resLoginTime time.Time
	resLoginTime, err = time.Parse(time.RFC3339, existingLoginTime)
	if err != nil {
		return time.Time{}, fmt.Errorf("parse login time: %v", err)
	}
    if err := updateDB(); err != nil {
        fmt.Printf("Warning: Failed to sync database to cloud: %v\n", err)
    }
	fmt.Println("Successfully saved login time.")
    return resLoginTime, nil
}



func checkSessionValidity(teamID int) (bool, int, error) {
	if err := loadDB(); err != nil {
		return false, 0, fmt.Errorf("failed to load database: %v", err)
	}

	db, err := sql.Open("sqlite3", tmpDBPath)
	if err != nil {
		return false, 0, fmt.Errorf("open DB: %v", err)
	}
	defer db.Close()

	var loginTimeStr string
	err = db.QueryRow("SELECT login_time FROM teams WHERE team_id = ?", teamID).Scan(&loginTimeStr)
	if err != nil {
		return false, 0, fmt.Errorf("query login time: %v", err)
	}

	loginTime, err := time.Parse(time.RFC3339, loginTimeStr)
	if err != nil {
		return false, 0, fmt.Errorf("parse login time: %v", err)
	}

	now := time.Now()
	elapsed := now.Sub(loginTime)
	remaining := (3 * time.Hour) - elapsed

	if remaining <= 0 {
		return false, 0, nil
	}

	return true, int(remaining.Seconds()), nil
}
